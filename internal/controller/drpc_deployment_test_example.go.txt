package controllers_test

import (
	"context"
	"time"

	. "github.com/onsi/ginkgo/v2"
	. "github.com/onsi/gomega"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
	"k8s.io/apimachinery/pkg/types"
	"sigs.k8s.io/controller-runtime/pkg/client"

	rmn "github.com/ramendr/ramen/api/v1alpha1"
	"github.com/ramendr/ramen/internal/controller/testutils"
	plrv1 "github.com/stolostron/multicloud-operators-placementrule/pkg/apis/apps/v1"
	clrapiv1beta1 "open-cluster-management.io/api/cluster/v1beta1"
)

// PlacementType defines the type of placement mechanism
type PlacementType int

const (
	UsePlacementRule PlacementType = iota
	UsePlacementWithSubscription
	UsePlacementWithAppSet
)

// DeploymentTestCase defines a test case for initial deployment
type DeploymentTestCase struct {
	name            string
	placementType   PlacementType
	namespace       string
	placementName   string
	homeCluster     string
	drPolicyName    string
	expectedPhase   rmn.DRState
	setupFunc       func(context.Context, client.Client) error
	verifyFunc      func(context.Context, client.Client) error
}

var _ = Describe("DRPC Initial Deployment - Refactored Example", func() {
	var (
		ctx           context.Context
		testNamespace string
	)

	BeforeEach(func() {
		ctx = context.Background()
		testNamespace = testutils.DefaultDRPCNamespace

		// Setup test environment
		By("Creating test namespaces")
		for _, ns := range testutils.GetTestNamespaces() {
			Expect(testutils.CreateNamespace(ctx, k8sClient, ns)).To(Succeed())
		}

		By("Creating managed clusters")
		for _, cluster := range testutils.GetAsyncClusters() {
			Expect(k8sClient.Create(ctx, cluster)).To(Succeed())
		}

		By("Creating DR policy")
		policy := testutils.GetAsyncDRPolicy()
		Expect(k8sClient.Create(ctx, policy)).To(Succeed())
	})

	AfterEach(func() {
		By("Cleaning up test resources")
		// Cleanup logic here
	})

	// Table-driven tests for different placement types
	deploymentTests := []DeploymentTestCase{
		{
			name:          "Deploy with PlacementRule",
			placementType: UsePlacementRule,
			namespace:     testutils.DefaultDRPCNamespace,
			placementName: testutils.UserPlacementRuleName,
			homeCluster:   testutils.East1ManagedCluster,
			drPolicyName:  testutils.AsyncDRPolicyName,
			expectedPhase: rmn.Deployed,
			setupFunc: func(ctx context.Context, c client.Client) error {
				// Create PlacementRule
				plRule := &plrv1.PlacementRule{
					ObjectMeta: metav1.ObjectMeta{
						Name:      testutils.UserPlacementRuleName,
						Namespace: testutils.DefaultDRPCNamespace,
					},
					Spec: plrv1.PlacementRuleSpec{
						GenericPlacementFields: plrv1.GenericPlacementFields{
							Clusters: []plrv1.GenericClusterReference{
								{Name: testutils.East1ManagedCluster},
							},
						},
						SchedulerName: "ramen",
					},
				}
				return c.Create(ctx, plRule)
			},
		},
		{
			name:          "Deploy with Placement and Subscription",
			placementType: UsePlacementWithSubscription,
			namespace:     testutils.DefaultDRPCNamespace,
			placementName: testutils.UserPlacementName,
			homeCluster:   testutils.East1ManagedCluster,
			drPolicyName:  testutils.AsyncDRPolicyName,
			expectedPhase: rmn.Deployed,
			setupFunc: func(ctx context.Context, c client.Client) error {
				// Create Placement
				placement := &clrapiv1beta1.Placement{
					ObjectMeta: metav1.ObjectMeta{
						Name:      testutils.UserPlacementName,
						Namespace: testutils.DefaultDRPCNamespace,
					},
					Spec: clrapiv1beta1.PlacementSpec{
						ClusterSets: []string{"default"},
					},
				}
				return c.Create(ctx, placement)
			},
		},
		{
			name:          "Deploy with Placement and ApplicationSet",
			placementType: UsePlacementWithAppSet,
			namespace:     testutils.DefaultDRPCNamespace,
			placementName: testutils.UserPlacementName,
			homeCluster:   testutils.East1ManagedCluster,
			drPolicyName:  testutils.AsyncDRPolicyName,
			expectedPhase: rmn.Deployed,
			setupFunc: func(ctx context.Context, c client.Client) error {
				// Create Placement
				placement := &clrapiv1beta1.Placement{
					ObjectMeta: metav1.ObjectMeta{
						Name:      testutils.UserPlacementName,
						Namespace: testutils.DefaultDRPCNamespace,
					},
					Spec: clrapiv1beta1.PlacementSpec{
						ClusterSets: []string{"default"},
					},
				}
				if err := c.Create(ctx, placement); err != nil {
					return err
				}

				// Create ApplicationSet
				appSet := testutils.GetDefaultApplicationSet()
				return c.Create(ctx, appSet)
			},
		},
	}

	// Run all deployment test cases
	for _, tc := range deploymentTests {
		tc := tc // Capture range variable
		Context(tc.name, func() {
			var drpc *rmn.DRPlacementControl

			It("Should setup placement successfully", func() {
				if tc.setupFunc != nil {
					Expect(tc.setupFunc(ctx, k8sClient)).To(Succeed())
				}
			})

			It("Should create DRPC successfully", func() {
				By("Creating DRPC using builder pattern")
				drpc = testutils.NewDRPCBuilder(testutils.DRPCCommonName, tc.namespace).
					WithPlacementRef(tc.placementName, getPlacementKind(tc.placementType)).
					WithDRPolicyRef(tc.drPolicyName).
					WithPreferredCluster(tc.homeCluster).
					WithPVCSelector(metav1.LabelSelector{
						MatchLabels: map[string]string{"appclass": "gold"},
					}).
					Build()

				Expect(k8sClient.Create(ctx, drpc)).To(Succeed())
			})

			It("Should reach Deployed phase", func() {
				By("Waiting for DRPC to be deployed")
				testutils.WaitForDRPCPhase(ctx, k8sClient,
					testutils.DRPCCommonName,
					tc.namespace,
					tc.expectedPhase,
					time.Minute*2)
			})

			It("Should have correct status conditions", func() {
				By("Verifying Available condition")
				Expect(testutils.VerifyDRPCCondition(ctx, k8sClient,
					testutils.DRPCCommonName,
					tc.namespace,
					rmn.ConditionAvailable,
					metav1.ConditionTrue)).To(Succeed())

				By("Verifying Protected condition")
				Expect(testutils.VerifyDRPCCondition(ctx, k8sClient,
					testutils.DRPCCommonName,
					tc.namespace,
					rmn.ConditionProtected,
					metav1.ConditionTrue)).To(Succeed())
			})

			It("Should create VRG ManifestWork on home cluster", func() {
				By("Checking ManifestWork count")
				count, err := testutils.GetManifestWorkCount(ctx, k8sClient, tc.homeCluster)
				Expect(err).ToNot(HaveOccurred())
				Expect(count).To(BeNumerically(">=", 1))
			})

			if tc.verifyFunc != nil {
				It("Should pass custom verification", func() {
					Expect(tc.verifyFunc(ctx, k8sClient)).To(Succeed())
				})
			}
		})
	}
})

// Helper function to get placement kind based on type
func getPlacementKind(placementType PlacementType) string {
	switch placementType {
	case UsePlacementRule:
		return "PlacementRule"
	case UsePlacementWithSubscription, UsePlacementWithAppSet:
		return "Placement"
	default:
		return "Placement"
	}
}

// Example of a focused test for a specific scenario
var _ = Describe("DRPC Deployment Edge Cases", func() {
	Context("When DRPolicy is missing", func() {
		It("Should not add finalizer to DRPC", func() {
			ctx := context.Background()

			By("Creating DRPC with non-existent policy")
			drpc := testutils.NewDRPCBuilder("test-drpc", testutils.DefaultDRPCNamespace).
				WithPlacementRef("test-placement", "Placement").
				WithDRPolicyRef("non-existent-policy").
				WithPreferredCluster(testutils.East1ManagedCluster).
				Build()

			Expect(k8sClient.Create(ctx, drpc)).To(Succeed())

			By("Verifying finalizer is not added")
			Eventually(func() bool {
				updatedDRPC := &rmn.DRPlacementControl{}
				err := k8sClient.Get(ctx, types.NamespacedName{
					Name:      drpc.Name,
					Namespace: drpc.Namespace,
				}, updatedDRPC)
				if err != nil {
					return false
				}
				return len(updatedDRPC.Finalizers) == 0
			}, time.Second*30, time.Second).Should(BeTrue())
		})
	})

	Context("When preferred cluster is not in DRPolicy", func() {
		It("Should report error in status", func() {
			ctx := context.Background()

			By("Creating DRPolicy with specific clusters")
			policy := testutils.NewDRPolicyBuilder("test-policy").
				WithClusters(testutils.East1ManagedCluster, testutils.West1ManagedCluster).
				WithSchedulingInterval("1h").
				Build()
			Expect(k8sClient.Create(ctx, policy)).To(Succeed())

			By("Creating DRPC with cluster not in policy")
			drpc := testutils.NewDRPCBuilder("test-drpc-2", testutils.DefaultDRPCNamespace).
				WithPlacementRef("test-placement", "Placement").
				WithDRPolicyRef("test-policy").
				WithPreferredCluster(testutils.East2ManagedCluster). // Not in policy
				Build()

			Expect(k8sClient.Create(ctx, drpc)).To(Succeed())

			By("Verifying error condition")
			Eventually(func() bool {
				updatedDRPC := &rmn.DRPlacementControl{}
				err := k8sClient.Get(ctx, types.NamespacedName{
					Name:      drpc.Name,
					Namespace: drpc.Namespace,
				}, updatedDRPC)
				if err != nil {
					return false
				}
				_, condition := testutils.GetDRPCCondition(&updatedDRPC.Status, rmn.ConditionAvailable)
				return condition != nil && condition.Status == metav1.ConditionFalse
			}, time.Second*30, time.Second).Should(BeTrue())
		})
	})
})
